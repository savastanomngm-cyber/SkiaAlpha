  $("#main").innerHTML=`<div class="wrap"><div class="card"><h1>💼 Open Positions — Lifecycle Monitor</h1><div class="sub">direction-aware stops · trailing exits · thesis decay · time stops</div><div id="out"><span class="loader"></span></div></div></div>`;
  try{
    const d=await api("/api/positions");const rows=d.positions||[];
    if(!rows.length){$("#out").innerHTML=`<div class="panel hold">No open positions. Run the pipeline — approved BUY/SELL signals auto-open tracked positions here.</div>`;return;}
    const pref=["ticker","direction","entry","entry_price","last","last_price","pnl_pct","shares","qty","days_held","stop","trail_stop","binding_constraint","status","opened"];
    const cols=[...new Set([...pref.filter(k=>rows.some(r=>r[k]!=null)),...Object.keys(rows[0])])];
    const pretty=k=>k.replace(/_/g," ").replace(/\b\w/g,c=>c.toUpperCase());
    let html=`<div class="tbl"><table><tr>${cols.map(c=>`<th class="${/pnl|price|last|entry|shares|qty|days|stop/i.test(c)?"r":""}">${pretty(c)}</th>`).join("")}</tr>`;
    rows.forEach(r=>{html+=`<tr class="clickable" data-ticker="${r.ticker||""}">`+cols.map(c=>{
      let v=r[c];if(v==null)v="—";
      if(/pnl/.test(c)&&typeof r[c]==="number")return `<td class="r">${fmtPct(r[c])}</td>`;
      if(typeof r[c]==="number")return `<td class="r">${num(r[c]).toLocaleString()}</td>`;
      if(/direction/i.test(c))return `<td class="c"><span class="sig ${String(v).toLowerCase()==="buy"||String(v).toLowerCase()==="long"?"buy":"sell"}">${v}</span></td>`;
      return `<td>${v}</td>`;}).join("")+"</tr>";});
    $("#out").innerHTML=html+"</table></div>";
  }catch(e){$("#out").innerHTML=`<div class="panel sell">Error: ${e.message}</div>`;}
}
